4. Add the Export Function (JS.html):
At the very bottom of the file, add this new function. It extracts only the requested 4 columns and natively triggers a CSV file download (which opens perfectly in Excel):

function exportRentReport() {
  if (!currentFilteredReportData || currentFilteredReportData.length === 0) {
    alert("No data to export!");
    return;
  }

  // Create CSV Headers
  let csvContent = "data:text/csv;charset=utf-8,Date,Name,Amount,MOP\n";

  // Loop through data and extract only the 4 required columns
  currentFilteredReportData.forEach(r => {
    // r.entity holds the Tenant Name
    let row = `"${r.date}","${r.entity}","${r.amount}","${r.mop || '-'}"`;
    csvContent += row + "\n";
  });

  // Trigger automatic download
  const encodedUri = encodeURI(csvContent);
  const link = document.createElement("a");
  link.setAttribute("href", encodedUri);
  link.setAttribute("download", "Rent_Report.csv");
  document.body.appendChild(link);
  link.click();
  document.body.removeChild(link);
}
